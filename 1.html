ConnectApp — Fullstendig flytoversikt

Hvert funksjonskall, tilstandsoppdatering, Firestore-skriving og re-rendring — fra sideinnlasting til arrangementsletting — sporet linje for linje fra de faktiske kildefilene. Ingen generaliseringer.

Denne appen er et vanillje-JS studentbolig-dashbord. Det er ingen rammeverk-ruter. Tilstand lever i lukninger. Nettleseren laster skript i rekkefølge og hvert modul publiserer seg selv på window.connectXxx-globals.

async function handleSubmit() {

  const isCommonRoom       = state.place === 'cr';
  const audienceMode       = String(state.audienceMode || 'anyone');
  const extraInviteTargets = audienceMode === 'anyone' ? [] : state.extraInviteTargets;

  if (isCommonRoom) {
    await refreshBusyDates();    // re-sjekk RETT FØR skriving
    syncDateConflict();

  }

  eventFormUiApi.setSubmitLoading({
    buttonElement: g('btn-submit'),
    label: isCommonRoom ? 'Booking...' : 'Creating...'
  });

  const result = await eventDataApi.createEventWithOptionalRoomBooking({
    db, serverTimestamp,

      name:           g('ev-name').value.trim(),
      description:    g('ev-desc').value.trim(),

      audienceMode,
      extraInviteTargets

    actorProfile: profile
  });

async function createEventWithOptionalRoomBooking(options) {
  const useCommonRoom = Boolean(formValues.useCommonRoom);

  // 1. Valider tittel (maks 25 tegn)
  const eventTitle = validateEventTitle(formValues.name);

  // 2. Bygg publikum/synlighetsfelter
  const audienceFields = buildAudienceFields({
    audienceMode:       formValues.audienceMode,
    actorProfile:       options.actorProfile,
    extraInviteTargets: formValues.extraInviteTargets
  });

  // 3. Ny auto-id-referanse (ingen skriving ennå)
  const eventRef = db.collection('events').doc();

  // 4. Sett sammen arrangementsdokumentet
  const eventData = {
    name: eventTitle, description: formValues.description.trim(),
    eventDate: formValues.eventDate, startTime: formValues.startTime,
    endTime: formValues.endTime || '', useCommonRoom,
    location: useCommonRoom ? 'Common Room' : (formValues.location || null),
    audienceMode: audienceFields.audienceMode,
    audienceScopeValue: audienceFields.audienceScopeValue,
    extraInviteUids: audienceFields.extraInviteUids,
    visibilityTargets: audienceFields.visibilityTargets,
    isPrivate: audienceFields.isPrivate,
    status: useCommonRoom ? 'pending' : 'visible',  // ← CR starter som pending
    createdByUid: actor.uid, createdByName: actor.displayName,
    createdAt: serverTimestamp()
  };

  const batch = db.batch();

  // 5. Fellesrom: opprett koblet bookingforespørsel
  if (useCommonRoom) {
    const bookingRequestRef = db.collection('bookingRequests').doc();
    eventData.bookingRequestId = bookingRequestRef.id;
    batch.set(bookingRequestRef, {
      eventId: eventRef.id, eventName: eventData.name,
      eventDate: eventData.eventDate, startTime: eventData.startTime,
      endTime: eventData.endTime, isPrivate: eventData.isPrivate,
      requestedByUid: actor.uid, requesterName: actor.displayName,
      status: 'pending', createdAt: serverTimestamp()
    });
  }

  // 6. Skriv arrangementsdokumentet
  batch.set(eventRef, eventData);

  // 7. Skriv invitasjonsdokumenter for hvert extraInviteTarget
  await syncEventInviteDocuments({
    db, batch, eventId: eventRef.id, eventData,
    existingInvites: [], serverTimestamp
  });

  // 8. Legg automatisk til oppretteren som deltaker
  batch.set(eventRef.collection('attendees').doc(actor.uid), {
    uid: actor.uid, displayName: actor.displayName, joinedAt: serverTimestamp()
  });

  await batch.commit();   // ← ÉN atomisk round-trip til Firestore

  return { eventId: eventRef.id, event: eventData };
}

async function deleteEvent(options) {
  const eventRef      = db.collection('events').doc(options.eventId);
  const eventSnapshot = await eventRef.get();
  const eventData     = eventSnapshot.exists ? eventSnapshot.data() : {};

  // 1. Last alle deltaker-underdokumenter
  const attendeesSnapshot = await eventRef.collection('attendees').get();

  // 2. Last alle invitasjonsdokumenter for dette arrangementet
  const inviteRecords = await loadEventInviteRecordsFromEvent({
    db, eventId: options.eventId,
    eventData: { id: options.eventId, ...eventData }
  });

  // 3. Slett deltakere i grupper på 450 (Firestore-grense per batch)
  for (let i = 0; i < attendeesSnapshot.docs.length; i += 450) {
    const attendeeBatch = db.batch();
    attendeesSnapshot.docs.slice(i, i + 450).forEach(doc => {
      attendeeBatch.delete(doc.ref);
    });
    await attendeeBatch.commit();
  }

  // 4. Hoved-oppryddingsbatch
  const batch = db.batch();

  if (eventData.bookingRequestId) {
    batch.delete(
      db.collection('bookingRequests').doc(eventData.bookingRequestId)
    );
  }

  inviteRecords.forEach(r => { if (r.ref) batch.delete(r.ref); });

  batch.delete(eventRef);   // sletter selve arrangementsdokumentet

  await batch.commit();

  return { eventId: options.eventId, deleted: true };
}

async function updateEventStatus(options) {
  const eventRef      = db.collection('events').doc(options.eventId);
  const eventSnapshot = await eventRef.get();

  const updateData = { status: options.status, updatedAt: options.serverTimestamp() };

  if (options.status === 'canceled' && options.cancellationReason) {
    updateData.cancellationReason = options.cancellationReason;
  }

  const batch = db.batch();
  batch.update(eventRef, updateData);  // ← status → 'canceled'

  // Bygg normalisert tilstand ETTER oppdateringen slik at invitasjonssynken
  // bruker avlyst-status ved skriving av invitasjonsdokumentfelter.
  const currentEvent = normalizeEventRecord({
    id: eventSnapshot.id, ...eventSnapshot.data(), ...updateData
  });

  const existingInvites = await loadEventInviteRecordsFromEvent({
    db, eventId: options.eventId,
    eventData: { id: eventSnapshot.id, ...eventSnapshot.data() }
  });

  // Oppdater alle invitasjonsdokumenter: eventStatus → 'canceled'
  await syncEventInviteDocuments({
    db, batch, eventId: options.eventId,
    eventData: currentEvent,
    existingInvites, serverTimestamp: options.serverTimestamp
  });

  await batch.commit();

  return { eventId: options.eventId, status: options.status };
}
Why this matters: Uten å synkronisere invitasjonsdokumentene ville en beboer som ble direkte invitert fortsatt se «ventende invitasjon»-varsling etter avlysing. eventStatus-feltet på invitasjonsdokumentet er det varslingsfilteringen sjekker mot.
